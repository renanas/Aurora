package br.com.aurora.auth.application;

import java.time.Clock;
import java.time.Duration;
import java.time.Instant;

/**
 * Limite de tentativas de login: depois de {@code maxFailures} falhas seguidas, o login fica bloqueado
 * por {@code lockDuration}. Falhas mais antigas que {@code lockDuration} são esquecidas.
 *
 * O bloqueio é global (não por IP): o sistema tem uma única usuária e, atrás de um proxy, todos os
 * IPs pareceriam o mesmo. Custo aceito: quem insistir de fora pode travar o login dela por alguns
 * minutos; o segundo fator (TOTP) é o passo seguinte para endurecer isso.
 *
 * Estado em memória: reiniciar a aplicação zera o contador. Classe pura (sem Spring) para ser
 * testada com um relógio controlado.
 */
public class LoginAttemptLimiter {

    private final Clock clock;
    private final int maxFailures;
    private final Duration lockDuration;

    private int failures;
    private Instant lastFailureAt;
    private Instant lockedUntil;

    public LoginAttemptLimiter(Clock clock, int maxFailures, Duration lockDuration) {
        if (maxFailures < 1) {
            throw new IllegalArgumentException("maxFailures deve ser pelo menos 1");
        }
        if (lockDuration.isZero() || lockDuration.isNegative()) {
            throw new IllegalArgumentException("lockDuration deve ser positiva");
        }
        this.clock = clock;
        this.maxFailures = maxFailures;
        this.lockDuration = lockDuration;
    }

    /** Tempo que ainda falta de bloqueio; {@link Duration#ZERO} se o login está liberado. */
    public synchronized Duration remainingLock() {
        if (lockedUntil == null) {
            return Duration.ZERO;
        }
        Instant now = clock.instant();
        if (!now.isBefore(lockedUntil)) {
            lockedUntil = null;
            return Duration.ZERO;
        }
        return Duration.between(now, lockedUntil);
    }

    public synchronized void recordFailure() {
        if (!remainingLock().isZero()) {
            return; // já bloqueado: não estende o bloqueio
        }
        Instant now = clock.instant();
        if (lastFailureAt != null && Duration.between(lastFailureAt, now).compareTo(lockDuration) >= 0) {
            failures = 0;
        }
        failures++;
        lastFailureAt = now;
        if (failures >= maxFailures) {
            lockedUntil = now.plus(lockDuration);
            failures = 0;
        }
    }

    /** Login bem-sucedido (ou reinício do estado): zera contador e bloqueio. */
    public synchronized void recordSuccess() {
        failures = 0;
        lastFailureAt = null;
        lockedUntil = null;
    }
}

package br.com.aurora.auth.web;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import org.springframework.security.web.csrf.CsrfToken;
import org.springframework.web.filter.OncePerRequestFilter;

/**
 * O Spring Security só gera o token CSRF quando alguém o lê. Como o frontend é uma SPA e nunca
 * renderiza o token, este filtro o lê em toda requisição para garantir que o cookie XSRF-TOKEN
 * seja enviado (inclusive na primeira resposta, que é um 401 de GET /api/auth/me).
 * Não é um bean: é criado dentro da SecurityConfig para não ser registrado duas vezes.
 */
final class CsrfCookieFilter extends OncePerRequestFilter {

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain chain)
            throws ServletException, IOException {
        CsrfToken token = (CsrfToken) request.getAttribute(CsrfToken.class.getName());
        if (token != null) {
            token.getToken(); // força a gravação do cookie
        }
        chain.doFilter(request, response);
    }
}

package br.com.aurora.auth.web;

import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.nio.charset.StandardCharsets;

/**
 * Escreve erros no mesmo formato do GlobalExceptionHandler (Problem Details com "code" estável)
 * para as respostas geradas pelos filtros de segurança, que ficam fora do Spring MVC.
 *
 * Todos os argumentos são constantes do código (sem aspas, sem dado do usuário): por isso a
 * montagem do JSON é feita à mão e nada da requisição é ecoado.
 */
final class JsonProblemWriter {

    private JsonProblemWriter() {
    }

    static void write(HttpServletResponse response, int status, String title, String code, String detail)
            throws IOException {
        String body = "{\"type\":\"about:blank\",\"title\":\"" + title + "\",\"status\":" + status
                + ",\"detail\":\"" + detail + "\",\"code\":\"" + code + "\"}";
        byte[] bytes = body.getBytes(StandardCharsets.UTF_8);
        response.setStatus(status);
        response.setContentType("application/problem+json");
        response.setCharacterEncoding("UTF-8");
        response.setContentLength(bytes.length);
        response.getOutputStream().write(bytes);
    }
}

package br.com.aurora.auth.web;

import br.com.aurora.auth.application.LoginAttemptLimiter;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.time.Duration;
import org.springframework.web.filter.OncePerRequestFilter;

/**
 * Recusa tentativas de login (429) enquanto o limite de tentativas estiver bloqueando, sem nem
 * verificar a senha. Fica antes do filtro de login do Spring Security.
 * Não é um bean: é criado dentro da SecurityConfig para não ser registrado duas vezes.
 */
final class LoginThrottleFilter extends OncePerRequestFilter {

    static final String LOGIN_PATH = "/api/auth/login";

    private final LoginAttemptLimiter limiter;

    LoginThrottleFilter(LoginAttemptLimiter limiter) {
        this.limiter = limiter;
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response, FilterChain chain)
            throws ServletException, IOException {
        if ("POST".equals(request.getMethod()) && LOGIN_PATH.equals(request.getRequestURI())) {
            Duration remaining = limiter.remainingLock();
            if (!remaining.isZero()) {
                long seconds = (remaining.toMillis() + 999) / 1000;
                long minutes = Math.max(1, (seconds + 59) / 60);
                response.setHeader("Retry-After", String.valueOf(seconds));
                JsonProblemWriter.write(response, 429, "Muitas tentativas", "TOO_MANY_ATTEMPTS",
                        "Muitas tentativas de entrada. Tente novamente em " + minutes
                                + (minutes == 1 ? " minuto." : " minutos."));
                return;
            }
        }
        chain.doFilter(request, response);
    }
}

package br.com.aurora.auth.web;

import br.com.aurora.auth.application.LoginAttemptLimiter;
import java.time.Clock;
import java.time.Duration;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.http.HttpStatus;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.core.userdetails.User;
import org.springframework.security.core.userdetails.UserDetailsService;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.provisioning.InMemoryUserDetailsManager;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;
import org.springframework.security.web.authentication.logout.HttpStatusReturningLogoutSuccessHandler;
import org.springframework.security.web.authentication.www.BasicAuthenticationFilter;
import org.springframework.security.web.csrf.CookieCsrfTokenRepository;
import org.springframework.security.web.csrf.CsrfTokenRequestAttributeHandler;
import org.springframework.security.web.savedrequest.NullRequestCache;

/**
 * Autenticação do Aurora: usuária única, sessão por cookie, tudo no mesmo domínio (sem CORS).
 *
 *  - Todo /api/** exige login; só /actuator/health é público. O que não é /api nem health é negado
 *    (nada sai sem sessão; o deploy decide depois como servir o frontend).
 *  - Login: POST /api/auth/login com formulário (username, password) -> 204. Logout: POST
 *    /api/auth/logout -> 204. Erros no formato Problem Details com "code".
 *  - CSRF: cookie XSRF-TOKEN legível pelo JavaScript, devolvido no cabeçalho X-XSRF-TOKEN.
 *  - Cookie de sessão (JSESSIONID): HttpOnly, SameSite=Strict, Secure em produção (application.yml).
 *  - A usuária vem de aurora.auth.username / aurora.auth.password (variáveis de ambiente
 *    AURORA_AUTH_USERNAME / AURORA_AUTH_PASSWORD). Sem elas, ninguém consegue entrar.
 */
@Configuration
@EnableWebSecurity
public class SecurityConfig {

    private static final Logger log = LoggerFactory.getLogger(SecurityConfig.class);

    static final int MIN_PASSWORD_LENGTH = 12;
    static final int MAX_LOGIN_FAILURES = 5;
    static final Duration LOGIN_LOCK = Duration.ofMinutes(15);

    @Bean
    LoginAttemptLimiter loginAttemptLimiter() {
        return new LoginAttemptLimiter(Clock.systemUTC(), MAX_LOGIN_FAILURES, LOGIN_LOCK);
    }

    @Bean
    PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    /**
     * Único usuário, em memória. A senha das variáveis de ambiente é transformada em hash (BCrypt)
     * na subida. Sem usuário configurado a aplicação sobe, mas ninguém loga (falha fechada).
     */
    @Bean
    UserDetailsService userDetailsService(
            @Value("${aurora.auth.username:}") String username,
            @Value("${aurora.auth.password:}") String password,
            PasswordEncoder encoder) {
        if (username.isBlank() || password.isBlank()) {
            log.warn("Usuário de login não configurado (AURORA_AUTH_USERNAME / AURORA_AUTH_PASSWORD): "
                    + "ninguém conseguirá entrar.");
            return new InMemoryUserDetailsManager();
        }
        if (password.length() < MIN_PASSWORD_LENGTH) {
            throw new IllegalStateException(
                    "aurora.auth.password precisa ter pelo menos " + MIN_PASSWORD_LENGTH + " caracteres.");
        }
        return new InMemoryUserDetailsManager(
                User.withUsername(username).password(encoder.encode(password)).roles("USER").build());
    }

    @Bean
    SecurityFilterChain securityFilterChain(
            HttpSecurity http,
            LoginAttemptLimiter limiter,
            @Value("${server.servlet.session.cookie.secure:false}") boolean secureCookies) throws Exception {

        // Mesmos atributos do cookie de sessão: SameSite=Strict e Secure conforme o ambiente.
        CookieCsrfTokenRepository csrfRepository = CookieCsrfTokenRepository.withHttpOnlyFalse();
        csrfRepository.setCookieCustomizer(cookie -> cookie.sameSite("Strict").secure(secureCookies));

        http
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers("/actuator/health", "/actuator/health/**").permitAll()
                        .requestMatchers("/api/**").authenticated()
                        .anyRequest().denyAll())
                .csrf(csrf -> csrf
                        .csrfTokenRepository(csrfRepository)
                        // O token vai por cookie e cabeçalho, nunca dentro de uma resposta
                        // comprimida (cenário do BREACH): o tratador simples basta.
                        .csrfTokenRequestHandler(new CsrfTokenRequestAttributeHandler()))
                .addFilterAfter(new CsrfCookieFilter(), BasicAuthenticationFilter.class)
                .addFilterBefore(new LoginThrottleFilter(limiter), UsernamePasswordAuthenticationFilter.class)
                .formLogin(form -> form
                        .loginPage("/login") // rota do frontend; impede a página de login padrão do Spring
                        .loginProcessingUrl(LoginThrottleFilter.LOGIN_PATH)
                        .successHandler((request, response, authentication) -> {
                            limiter.recordSuccess();
                            response.setStatus(HttpStatus.NO_CONTENT.value());
                        })
                        .failureHandler((request, response, exception) -> {
                            limiter.recordFailure();
                            JsonProblemWriter.write(response, 401, "Credenciais inválidas",
                                    "BAD_CREDENTIALS", "Usuário ou senha incorretos.");
                        }))
                .logout(logout -> logout
                        .logoutUrl("/api/auth/logout")
                        .logoutSuccessHandler(new HttpStatusReturningLogoutSuccessHandler(HttpStatus.NO_CONTENT))
                        .deleteCookies("JSESSIONID"))
                .exceptionHandling(handling -> handling
                        .authenticationEntryPoint((request, response, exception) ->
                                JsonProblemWriter.write(response, 401, "Não autenticado",
                                        "UNAUTHENTICATED", "É preciso entrar para continuar."))
                        .accessDeniedHandler((request, response, exception) ->
                                JsonProblemWriter.write(response, 403, "Acesso negado", "FORBIDDEN",
                                        "A página ficou desatualizada. Atualize e tente de novo.")))
                // Sem sessão só para "lembrar a URL pedida": a SPA não precisa disso.
                .requestCache(cache -> cache.requestCache(new NullRequestCache()));

        return http.build();
    }
}

import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

import java.time.Clock;
import java.time.Duration;
import java.time.Instant;
import java.time.ZoneId;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.DisplayName;
import org.junit.jupiter.api.Test;

class LoginAttemptLimiterTest {

    /** Relógio que só anda quando o teste manda. */
    static class MutableClock extends Clock {
        private Instant now = Instant.parse("2026-10-17T12:00:00Z");

        void advance(Duration duration) {
            now = now.plus(duration);
        }

        @Override
        public ZoneId getZone() {
            return ZoneId.of("UTC");
        }

        @Override
        public Clock withZone(ZoneId zone) {
            return this;
        }

        @Override
        public Instant instant() {
            return now;
        }
    }

    MutableClock clock;
    LoginAttemptLimiter limiter;

    @BeforeEach
    void setUp() {
        clock = new MutableClock();
        limiter = new LoginAttemptLimiter(clock, 5, Duration.ofMinutes(15));
    }

    private void fail(int times) {
        for (int i = 0; i < times; i++) {
            limiter.recordFailure();
        }
    }

    @Test
    @DisplayName("começa liberado")
    void startsUnlocked() {
        assertThat(limiter.remainingLock()).isZero();
    }

    @Test
    @DisplayName("abaixo do limite continua liberado")
    void belowLimitStaysUnlocked() {
        fail(4);

        assertThat(limiter.remainingLock()).isZero();
    }

    @Test
    @DisplayName("na quinta falha seguida bloqueia pelo tempo configurado")
    void locksAfterMaxFailures() {
        fail(5);

        assertThat(limiter.remainingLock()).isEqualTo(Duration.ofMinutes(15));
    }

    @Test
    @DisplayName("o tempo de bloqueio diminui e o login volta depois dele")
    void lockExpires() {
        fail(5);

        clock.advance(Duration.ofMinutes(10));
        assertThat(limiter.remainingLock()).isEqualTo(Duration.ofMinutes(5));

        clock.advance(Duration.ofMinutes(5));
        assertThat(limiter.remainingLock()).isZero();
    }

    @Test
    @DisplayName("depois que o bloqueio acaba, a contagem recomeça do zero")
    void countRestartsAfterLock() {
        fail(5);
        clock.advance(Duration.ofMinutes(15));

        fail(4);

        assertThat(limiter.remainingLock()).isZero();
    }

    @Test
    @DisplayName("login bem-sucedido zera as falhas")
    void successResetsFailures() {
        fail(4);
        limiter.recordSuccess();

        fail(4);

        assertThat(limiter.remainingLock()).isZero();
    }

    @Test
    @DisplayName("falhas antigas (além da janela) são esquecidas")
    void oldFailuresAreForgotten() {
        fail(4);
        clock.advance(Duration.ofMinutes(15));

        fail(1);

        assertThat(limiter.remainingLock()).isZero();
    }

    @Test
    @DisplayName("falha durante o bloqueio não estende o bloqueio")
    void failureWhileLockedDoesNotExtend() {
        fail(5);
        clock.advance(Duration.ofMinutes(5));

        fail(3);

        assertThat(limiter.remainingLock()).isEqualTo(Duration.ofMinutes(10));
    }

    @Test
    @DisplayName("recordSuccess também libera um bloqueio (usado para reiniciar o estado)")
    void successClearsLock() {
        fail(5);

        limiter.recordSuccess();

        assertThat(limiter.remainingLock()).isZero();
    }

    @Test
    @DisplayName("rejeita configuração inválida")
    void rejectsInvalidConfig() {
        assertThatThrownBy(() -> new LoginAttemptLimiter(clock, 0, Duration.ofMinutes(1)))
                .isInstanceOf(IllegalArgumentException.class);
        assertThatThrownBy(() -> new LoginAttemptLimiter(clock, 3, Duration.ZERO))
                .isInstanceOf(IllegalArgumentException.class);
    }

/**

 * Os outros testes de API montam o MockMvc sem o Spring Security (de propósito: testam as regras

@SpringBootTest(properties = {
        "aurora.auth.username=psicologa",
        "aurora.auth.password=senha-de-teste-123"
})
@Import(TestcontainersConfig.class)
class AuthApiTest {

    private static final String USER = "psicologa";
    private static final String PASSWORD = "senha-de-teste-123";

    @Autowired
    WebApplicationContext context;
    @Autowired
    LoginAttemptLimiter limiter;

    MockMvc mvc;

    @BeforeEach
    void setUp() {
        mvc = MockMvcBuilders.webAppContextSetup(context).apply(springSecurity()).build();
        limiter.recordSuccess(); // o limitador é um bean compartilhado entre os testes: reinicia
    }

    // ---------- helpers ----------

    /** Pede qualquer rota e lê o token CSRF que o servidor entrega no cookie XSRF-TOKEN. */
    private String csrfToken() throws Exception {
        MvcResult result = mvc.perform(get("/api/auth/me")).andReturn();
        List<String> cookies = result.getResponse().getHeaders("Set-Cookie");
        String prefix = "XSRF-TOKEN=";
        return cookies.stream()
                .filter(c -> c.startsWith(prefix))
                .map(c -> {
                    int end = c.indexOf(';');
                    return c.substring(prefix.length(), end < 0 ? c.length() : end);
                })
                .findFirst()
                .orElseThrow(() -> new AssertionError("o servidor não enviou o cookie XSRF-TOKEN"));
    }

    private ResultActions login(String username, String password, String token) throws Exception {
        return mvc.perform(post("/api/auth/login")
                .contentType(MediaType.APPLICATION_FORM_URLENCODED)
                .param("username", username)
                .param("password", password)
                .cookie(new Cookie("XSRF-TOKEN", token))
                .header("X-XSRF-TOKEN", token));
    }

    private MockHttpSession loggedInSession(String token) throws Exception {
        MvcResult result = login(USER, PASSWORD, token).andExpect(status().isNoContent()).andReturn();
        return (MockHttpSession) result.getRequest().getSession(false);
    }

    // ---------- sem login ----------

    @Test
    @DisplayName("sem login, a API responde 401 com código estável")
    void apiRequiresLogin() throws Exception {
        mvc.perform(get("/api/patients"))
                .andExpect(status().isUnauthorized())
                .andExpect(jsonPath("$.code").value("UNAUTHENTICATED"));
    }

    @Test
    @DisplayName("a primeira resposta já entrega o cookie do token CSRF")
    void csrfCookieIsIssued() throws Exception {
        MvcResult result = mvc.perform(get("/api/auth/me")).andExpect(status().isUnauthorized()).andReturn();

        assertThat(result.getResponse().getHeaders("Set-Cookie"))
                .anyMatch(c -> c.startsWith("XSRF-TOKEN="));
    }

    @Test
    @DisplayName("o health check continua público")
    void healthIsPublic() throws Exception {
        MvcResult result = mvc.perform(get("/actuator/health")).andReturn();

        assertThat(result.getResponse().getStatus()).isNotEqualTo(401);
    }

    @Test
    @DisplayName("login correto abre a sessão e /me responde com o usuário")
    void loginOpensSession() throws Exception {
        MockHttpSession session = loggedInSession(csrfToken());

        mvc.perform(get("/api/auth/me").session(session))
                .andExpect(status().isOk())
                .andExpect(jsonPath("$.username").value(USER));
        mvc.perform(get("/api/patients").session(session))
                .andExpect(status().isOk());
    }

                .andExpect(jsonPath("$.code").value("BAD_CREDENTIALS"));
    }

    @Test
    @DisplayName("login sem token CSRF é recusado")
    void loginWithoutCsrfIsRejected() throws Exception {
        mvc.perform(post("/api/auth/login")

                        .param("username", USER)
                        .param("password", PASSWORD))
